## 案例一：3D 产品展示（电商）

**涉及**：`09` 材质环境贴图、`11` OrbitControls、`17` GLTFLoader、`13` Bloom

```js
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
import { GLTFLoader } from 'three/addons/loaders/GLTFLoader.js';
import { RGBELoader } from 'three/addons/loaders/RGBELoader.js';
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';
import { OutputPass } from 'three/addons/postprocessing/OutputPass.js';

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(45, innerWidth / innerHeight, 0.1, 100);
camera.position.set(0, 0.5, 4);

const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.1;
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
document.getElementById('app').appendChild(renderer.domElement);

// ---- 环境光照（PBR 关键）----
new RGBELoader().load('/hdr/studio.hdr', (tex) => {
  tex.mapping = THREE.EquirectangularReflectionMapping;
  const pmrem = new THREE.PMREMGenerator(renderer);
  scene.environment = pmrem.fromEquirectangular(tex).texture;
  tex.dispose();
  pmrem.dispose();
});

// ---- 控制器：限制视角，禁止到产品下方 ----
const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.08;
controls.enablePan = false;
controls.minDistance = 2;
controls.maxDistance = 6;
controls.minPolarAngle = Math.PI / 6;
controls.maxPolarAngle = Math.PI / 2;
controls.autoRotate = true;
controls.autoRotateSpeed = 1.0;
controls.addEventListener('start', () => controls.autoRotate = false);  // 用户操作后停止自转

// ---- 加载产品模型 ----
let product;
new GLTFLoader().load('/models/product.glb', (gltf) => {
  product = gltf.scene;

  // 归一化尺寸 + 居中
  const box = new THREE.Box3().setFromObject(product);
  const size = box.getSize(new THREE.Vector3());
  const center = box.getCenter(new THREE.Vector3());
  product.position.sub(center);
  product.scale.setScalar(2 / Math.max(size.x, size.y, size.z));

  product.traverse((c) => {
    if (c.isMesh) { c.castShadow = true; c.receiveShadow = true; }
  });
  scene.add(product);
  hideLoading();
});

// ---- 地面接触阴影 ----
const ground = new THREE.Mesh(
  new THREE.PlaneGeometry(20, 20),
  new THREE.ShadowMaterial({ opacity: 0.25 })
);
ground.rotation.x = -Math.PI / 2;
ground.position.y = -1;
ground.receiveShadow = true;
scene.add(ground);

const key = new THREE.DirectionalLight(0xffffff, 2);
key.position.set(4, 6, 3);
key.castShadow = true;
key.shadow.mapSize.set(2048, 2048);
key.shadow.bias = -0.0005;
scene.add(key);

// ---- 后期：轻微泛光 ----
const composer = new EffectComposer(renderer);
composer.addPass(new RenderPass(scene, camera));
composer.addPass(new UnrealBloomPass(new THREE.Vector2(innerWidth, innerHeight), 0.3, 0.6, 0.9));
composer.addPass(new OutputPass());

// ---- 换色/换材质（业务交互）----
function setColor(hex) {
  product?.traverse((c) => {
    if (c.isMesh && c.material.name === 'body') c.material.color.set(hex);
  });
}

// ---- 渲染 ----
const clock = new THREE.Clock();
renderer.setAnimationLoop(() => {
  controls.update(clock.getDelta());
  composer.render();
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
  composer.setSize(innerWidth, innerHeight);
});
```

**要点**：`scene.environment` 让 PBR 真实；限制相机角度避免看到模型底部穿帮；`ShadowMaterial` 做只显示阴影的地面（比实体地面干净）。

## 案例二：数据可视化大屏（地球 + 飞线）

**涉及**：`08` 自定义几何、`09` 贴图、`18` 粒子、`13` Bloom、`11` Raycaster

```js
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(45, innerWidth / innerHeight, 0.1, 1000);
camera.position.set(0, 3, 8);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
renderer.toneMapping = THREE.ACESFilmicToneMapping;
document.body.appendChild(renderer.domElement);

const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.minDistance = 4;
controls.maxDistance = 20;

// ============ 地球 ============
const earth = new THREE.Mesh(
  new THREE.SphereGeometry(2, 64, 64),
  new THREE.MeshStandardMaterial({
    map: loadSRGB('/earth/color.jpg'),
    emissiveMap: loadSRGB('/earth/night-lights.png'),
    emissive: 0xffffff,
    emissiveIntensity: 0.6,
    roughnessMap: loadLinear('/earth/roughness.jpg'),
    metalness: 0.1,
    roughness: 1,
  })
);
scene.add(earth);

function loadSRGB(url) {
  const t = new THREE.TextureLoader().load(url);
  t.colorSpace = THREE.SRGBColorSpace;
  return t;
}
function loadLinear(url) { return new THREE.TextureLoader().load(url); }

// ============ 大气辉光（背面 shader）============
const glow = new THREE.Mesh(
  new THREE.SphereGeometry(2.15, 64, 64),
  new THREE.ShaderMaterial({
    vertexShader: `
      varying vec3 vNormal;
      void main() {
        vNormal = normalize(normalMatrix * normal);
        gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
      }
    `,
    fragmentShader: `
      varying vec3 vNormal;
      void main() {
        // 边缘更亮（菲涅尔效果）
        float intensity = pow(0.7 - dot(vNormal, vec3(0, 0, 1.0)), 3.0);
        gl_FragColor = vec4(0.3, 0.6, 1.0, 1.0) * intensity;
      }
    `,
    side: THREE.BackSide,
    transparent: true,
    blending: THREE.AdditiveBlending,
    depthWrite: false,
  })
);
scene.add(glow);

// ============ 外围星点 ============
const starCount = 3000;
const starGeo = new THREE.BufferGeometry();
const starPos = new Float32Array(starCount * 3);
for (let i = 0; i < starCount; i++) {
  const r = 30 + Math.random() * 40;
  const theta = Math.random() * Math.PI * 2;
  const phi = Math.acos(Math.random() * 2 - 1);
  starPos[i * 3]     = r * Math.sin(phi) * Math.cos(theta);
  starPos[i * 3 + 1] = r * Math.cos(phi);
  starPos[i * 3 + 2] = r * Math.sin(phi) * Math.sin(theta);
}
starGeo.setAttribute('position', new THREE.BufferAttribute(starPos, 3));
scene.add(new THREE.Points(starGeo, new THREE.PointsMaterial({
  size: 0.15, color: 0xffffff, sizeAttenuation: true, transparent: true,
})));

// ============ 飞线（北京 → 各城市）============
// 经纬度转球面坐标
function latLngToVec3(lat, lng, radius) {
  const phi = (90 - lat) * Math.PI / 180;
  const theta = (lng + 180) * Math.PI / 180;
  return new THREE.Vector3(
    -radius * Math.sin(phi) * Math.cos(theta),
     radius * Math.cos(phi),
     radius * Math.sin(phi) * Math.sin(theta)
  );
}

const R = 2.02;
const beijing = { lat: 39.9, lng: 116.4 };
const cities = [
  { lat: 40.7, lng: -74.0 },   // 纽约
  { lat: 51.5, lng: -0.12 },   // 伦敦
  { lat: 35.7, lng: 139.7 },   // 东京
  { lat: -23.5, lng: -46.6 },  // 圣保罗
];

const flyingLines = [];
cities.forEach((c) => {
  const start = latLngToVec3(beijing.lat, beijing.lng, R);
  const end   = latLngToVec3(c.lat, c.lng, R);

  // 二次贝塞尔：中点抬高做弧线
  const mid = start.clone().add(end).multiplyScalar(0.5);
  const height = start.distanceTo(end) * 0.35;
  mid.normalize().multiplyScalar(R + height);

  const curve = new THREE.QuadraticBezierCurve3(start, mid, end);
  const points = curve.getPoints(60);

  const geo = new THREE.BufferGeometry().setFromPoints(points);
  const mat = new THREE.LineBasicMaterial({
    color: 0x00ffff, transparent: true, opacity: 0.8,
  });
  const line = new THREE.Line(geo, mat);
  scene.add(line);

  // 沿线飞行的光点
  const dot = new THREE.Mesh(
    new THREE.SphereGeometry(0.03, 8, 8),
    new THREE.MeshBasicMaterial({ color: 0x00ffff })
  );
  scene.add(dot);

  flyingLines.push({ curve, dot, progress: Math.random() });
});

// ============ 标记点 + 点击 ============
const markers = [];
const raycaster = new THREE.Raycaster();
const pointer = new THREE.Vector2();

cities.forEach((c, i) => {
  const pos = latLngToVec3(c.lat, c.lng, R);
  const marker = new THREE.Mesh(
    new THREE.SphereGeometry(0.06, 16, 16),
    new THREE.MeshBasicMaterial({ color: 0xff3355 })
  );
  marker.position.copy(pos);
  marker.userData.city = i;
  scene.add(marker);
  markers.push(marker);
});

addEventListener('click', (e) => {
  pointer.x = (e.clientX / innerWidth) * 2 - 1;
  pointer.y = -(e.clientY / innerHeight) * 2 + 1;
  raycaster.setFromCamera(pointer, camera);
  const hits = raycaster.intersectObjects(markers, false);
  if (hits.length) {
    console.log('点击城市索引:', hits[0].object.userData.city);
  }
});

// ============ 光照 & 渲染 ============
scene.add(new THREE.AmbientLight(0xffffff, 0.35));
const sun = new THREE.DirectionalLight(0xffffff, 2.2);
sun.position.set(5, 3, 5);
scene.add(sun);

const clock = new THREE.Clock();
renderer.setAnimationLoop(() => {
  const dt = clock.getDelta();
  const t = clock.getElapsedTime();

  earth.rotation.y += dt * 0.03;
  glow.rotation.y = earth.rotation.y;

  // 飞线光点沿曲线运动
  flyingLines.forEach(({ curve, dot, progress }) => {
    progress += dt * 0.25;
    if (progress > 1) progress -= 1;
    curve.getPointAt(progress, dot.position);
  });

  // 标记点呼吸
  const s = 1 + Math.sin(t * 3) * 0.25;
  markers.forEach((m) => m.scale.setScalar(s));

  controls.update();
  renderer.render(scene, camera);
});
```

**要点**：`QuadraticBezierCurve3` 做弧线飞线；`getPointAt(t)` 让光点沿线运动；BackSide 球 + 菲涅尔 shader 做大气辉光；经纬度转球面坐标是数据可视化的基础工具函数。

## 案例三：全景看房（HDR 天空盒 + 热点跳转）

**涉及**：`09` 环境贴图、`11` Raycaster、`12` 补间动画

```js
import * as THREE from 'three';
import { RGBELoader } from 'three/addons/loaders/RGBELoader.js';

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(75, innerWidth / innerHeight, 0.1, 1000);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
renderer.toneMapping = THREE.ACESFilmicToneMapping;
document.body.appendChild(renderer.domElement);

// ============ 房间数据（全景图 + 热点）============
const rooms = {
  living: {
    hdr: '/pano/living-room.hdr',
    hotspots: [
      { position: new THREE.Vector3(-4, -0.5, -3), target: 'bedroom', label: '卧室' },
      { position: new THREE.Vector3(4, -0.5, -2),  target: 'kitchen', label: '厨房' },
    ],
  },
  bedroom: {
    hdr: '/pano/bedroom.hdr',
    hotspots: [
      { position: new THREE.Vector3(0, -0.5, -4), target: 'living', label: '客厅' },
    ],
  },
  kitchen: {
    hdr: '/pano/kitchen.hdr',
    hotspots: [
      { position: new THREE.Vector3(0, -0.5, 4), target: 'living', label: '返回客厅' },
    ],
  },
};

// ============ 天空盒（球体贴全景图）============
const sphereGeo = new THREE.SphereGeometry(10, 64, 64);
sphereGeo.scale(-1, 1, 1);          // 翻转到内侧（关键！）
const panoMaterial = new THREE.MeshBasicMaterial();
const skybox = new THREE.Mesh(sphereGeo, panoMaterial);
scene.add(skybox);

// ============ 热点 ============
const hotspotGroup = new THREE.Group();
scene.add(hotspotGroup);

function createHotspot(position, label, target) {
  // 圆环 + 中心点
  const ring = new THREE.Mesh(
    new THREE.RingGeometry(0.12, 0.16, 32),
    new THREE.MeshBasicMaterial({ color: 0xffffff, side: THREE.DoubleSide, transparent: true })
  );
  ring.position.copy(position);
  ring.userData.target = target;
  ring.lookAt(camera.position);     // 面向相机

  hotspotGroup.add(ring);
  return ring;
}

function loadRoom(name) {
  const room = rooms[name];

  // 热度：清掉旧热点
  hotspotGroup.clear();

  new RGBELoader().load(room.hdr, (tex) => {
    tex.mapping = THREE.EquirectangularReflectionMapping;
    // 释放旧纹理
    if (panoMaterial.map) panoMaterial.map.dispose();
    panoMaterial.map = tex;
    panoMaterial.needsUpdate = true;
  });

  room.hotspots.forEach((h) => createHotspot(h.position, h.label, h.target));
}

// ============ 鼠标交互：悬停高亮 + 点击跳转 ============
const raycaster = new THREE.Raycaster();
const pointer = new THREE.Vector2();
let hovered = null;

renderer.domElement.addEventListener('pointermove', (e) => {
  pointer.x = (e.clientX / innerWidth) * 2 - 1;
  pointer.y = -(e.clientY / innerHeight) * 2 + 1;
});

renderer.domElement.addEventListener('click', () => {
  if (hovered) {
    fadeToBlack(() => loadRoom(hovered.userData.target));
  }
});

// 转场：淡黑再切换
const fadeOverlay = document.createElement('div');
fadeOverlay.style.cssText =
  'position:fixed;inset:0;background:#000;opacity:0;pointer-events:none;transition:opacity .4s;';
document.body.appendChild(fadeOverlay);

function fadeToBlack(callback) {
  fadeOverlay.style.opacity = '1';
  setTimeout(() => {
    callback();
    fadeOverlay.style.opacity = '0';
  }, 400);
}

// ============ 拖拽看四周（不用 OrbitControls，自己做）============
let isDragging = false, lon = 0, lat = 0, prevX = 0, prevY = 0;

renderer.domElement.addEventListener('pointerdown', (e) => {
  isDragging = true; prevX = e.clientX; prevY = e.clientY;
});
renderer.domElement.addEventListener('pointerup', () => isDragging = false);
renderer.domElement.addEventListener('pointermove', (e) => {
  if (!isDragging) return;
  lon += (e.clientX - prevX) * 0.15;
  lat += (e.clientY - prevY) * 0.15;
  lat = Math.max(-85, Math.min(85, lat));    // 限制俯仰
  prevX = e.clientX; prevY = e.clientY;
});

// 滚轮缩放（改 fov）
addEventListener('wheel', (e) => {
  camera.fov = Math.max(30, Math.min(90, camera.fov + e.deltaY * 0.05));
  camera.updateProjectionMatrix();
});

// ============ 渲染 ============
const clock = new THREE.Clock();
renderer.setAnimationLoop(() => {
  const t = clock.getElapsedTime();

  // 经纬度 → 相机朝向
  const phi = THREE.MathUtils.degToRad(90 - lat);
  const theta = THREE.MathUtils.degToRad(lon);
  camera.lookAt(
    10 * Math.sin(phi) * Math.cos(theta),
    10 * Math.cos(phi),
    10 * Math.sin(phi) * Math.sin(theta)
  );

  // 热点：面向相机 + 呼吸
  raycaster.setFromCamera(pointer, camera);
  const hits = raycaster.intersectObjects(hotspotGroup.children, false);
  hovered = hits.length ? hits[0].object : null;

  hotspotGroup.children.forEach((h) => {
    h.lookAt(camera.position);
    const active = h === hovered;
    h.scale.setScalar(active ? 1.3 + Math.sin(t * 4) * 0.1 : 1);
    h.material.opacity = active ? 1 : 0.75;
    document.body.style.cursor = active ? 'pointer' : 'auto';
  });

  renderer.render(scene, camera);
});

loadRoom('living');
```

**要点**：全景看房本质是**相机固定在球心、贴全景图的球翻转到内侧**（`scale(-1,1,1)`）；热点用 Raycaster 拾取；房间切换时**必须 `dispose()` 旧纹理**否则显存泄漏（见 `16`）。

## 通用项目清单

上线前检查：

| 项 | 检查 |
|----|------|
| 加载 | 有 loading 进度；模型用 DRACO/KTX2 压缩 |
| 性能 | `renderer.info.render.calls` 达标；像素比 ≤ 2 |
| 内存 | 切换场景 `dispose()` geometry/material/texture |
| 兼容 | 移动端触控；WebGL2 fallback |
| 体验 | resize 处理；错误兜底；降级方案 |
| 调试 | 保留 stats / helpers 但生产环境关闭 |

## 一句话总结

> 三个实战：产品展示（环境贴图 + 角度限制 + 接触阴影）、数据大屏（贝塞尔飞线 + 菲涅尔辉光 + 经纬度转换）、全景看房（内翻天空球 + 热点跳转 + 纹理释放）。核心都是把 08/09/11/13/16/17 的知识点组合，并注意资源释放与性能预算。
